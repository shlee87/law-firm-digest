import {
  getGeminiCallCount,

// Mock @google/genai to control generateContent behavior per test.

vi.mock('@google/genai', () => ({
  GoogleGenAI: vi.fn().mockImplementation(() => ({
    models: { generateContent: generateContentMock },
  })),

describe('src/summarize/gemini.ts module-level geminiCallCount (D-18)', () => {
  beforeEach(() => {
    resetGeminiCallCount();
    generateContentMock.mockReset();
    process.env.GEMINI_API_KEY = 'fake-key-for-test';
  });
  afterEach(() => {
    resetGeminiCallCount();
    delete process.env.GEMINI_API_KEY;
  });

  it('starts at 0', () => {
    expect(getGeminiCallCount()).toBe(0);

import { describe, it, expect, beforeEach, afterEach, vi } from 'vitest';
import { mkdtempSync, rmSync, readFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import * as path from 'node:path';
import { writeStepSummary } from '../../src/observability/summary.js';
import { Recorder } from '../../src/observability/recorder.js';
import type { FirmConfig } from '../../src/types.js';
import type { DataQualityMarker } from '../../src/pipeline/detectClusters.js';

describe('writeStepSummary [METRIC] geminiCallCount prepend (Phase 13 D-21/D-22)', () => {
  let dir: string;
  let summaryPath: string;
  let recorder: Recorder;
  const firms: FirmConfig[] = [];

  beforeEach(() => {
    dir = mkdtempSync(path.join(tmpdir(), 'step-summary-'));
    summaryPath = path.join(dir, 'summary.md');
    process.env.GITHUB_STEP_SUMMARY = summaryPath;
    recorder = new Recorder();
  });
  afterEach(() => {
    delete process.env.GITHUB_STEP_SUMMARY;
    rmSync(dir, { recursive: true, force: true });
  });

  it('prepends `[METRIC] geminiCallCount=N` as the first line when N > 0', async () => {
    await writeStepSummary(recorder, firms, [], 12);
    const out = readFileSync(summaryPath, 'utf8');
    expect(out.startsWith('[METRIC] geminiCallCount=12\n\n')).toBe(true);
  });

  it('emits `[METRIC] geminiCallCount=0` when 4th arg omitted (backwards-compat default)', async () => {
    await writeStepSummary(recorder, firms);
    const out = readFileSync(summaryPath, 'utf8');
    expect(out.startsWith('[METRIC] geminiCallCount=0\n\n')).toBe(true);
  });

  it('preserves marker ordering: [METRIC] → table → ## ⚠ Data Quality Warnings', async () => {
    const markers: DataQualityMarker[] = [
      { kind: 'cluster', firmId: 'bkl', count: 3 },
    ];
    await writeStepSummary(recorder, firms, markers, 7);
    const out = readFileSync(summaryPath, 'utf8');
    const metricIdx = out.indexOf('[METRIC] geminiCallCount=7');
    const warningsIdx = out.indexOf('## ⚠ Data Quality Warnings');
    expect(metricIdx).toBe(0);
    expect(warningsIdx).toBeGreaterThan(metricIdx);
  });

  it('no-op when GITHUB_STEP_SUMMARY env unset', async () => {
    delete process.env.GITHUB_STEP_SUMMARY;
    await expect(writeStepSummary(recorder, firms, [], 5)).resolves.toBeUndefined();
  });
});
